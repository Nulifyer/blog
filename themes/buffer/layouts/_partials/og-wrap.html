" " -}} {{- end -}} {{- end -}} {{- with $cur }}{{ $lines = $lines | append . }}{{ end -}} {{- if gt (len $lines) .lines -}} {{- $keep := first .lines $lines -}} {{- $last := printf "%s…" (index $keep (sub .lines 1)) -}} {{- $lines = (first (sub .lines 1) $keep) | append $last -}} {{- end -}} {{- $out := delimit $lines "\n" -}} {{- return $out -}}
