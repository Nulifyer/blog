{{- /* GitHub contribution calendar, read at build time from the public calendar page (no token needed). */ -}} {{- $user := site.Params.github.user -}} {{- $url := printf "https://github.com/users/%s/contributions" $user -}} {{- $html := "" -}} {{- with try (resources.GetRemote $url (dict "headers" (dict "User-Agent" "blog-builder (Hugo)"))) -}} {{- with .Err -}}{{- errorf "contributions: could not read %s: %s" $url . -}} {{- else with .Value -}}{{- $html = .Content -}}{{- end -}} {{- end -}} {{- /* Day counts come from the tooltips: for="contribution-day-component-ROW-COL" ... "3 contributions on ..." */ -}} {{- $counts := dict -}} {{- range findRESubmatch `for="(contribution-day-component-\d+-\d+)"[^>]*>\s*(No|\d+) contributions?` $html -}} {{- $n := 0 }}{{ if ne (index . 2) "No" }}{{ $n = int (index . 2) }}{{ end -}} {{- $counts = merge $counts (dict (index . 1) $n) -}} {{- end -}} {{- /* Day cells: id gives ROW (weekday, 0 = Sunday) and COL (week); data-level is GitHub's 0-4 shade. */ -}} {{- $cells := slice -}} {{- $weeks := 0 -}} {{- range findRE `]*class="ContributionCalendar-day"[^>]*>` $html -}} {{- $id := index (findRESubmatch `id="(contribution-day-component-(\d+)-(\d+))"` . 1) 0 -}} {{- $date := index (index (findRESubmatch `data-date="([\d-]+)"` . 1) 0) 1 -}} {{- $lvl := index (index (findRESubmatch `data-level="(\d)"` . 1) 0) 1 -}} {{- $col := int (index $id 3) -}} {{- if ge $col $weeks }}{{ $weeks = add $col 1 }}{{ end -}} {{- $cells = $cells | append (dict "row" (int (index $id 2)) "col" $col "date" (time.AsTime $date) "level" $lvl "count" (index $counts (index $id 1) | default 0)) -}} {{- end -}} {{- $total := 0 -}} {{- with findRESubmatch `([\d,]+)\s+contributions?\s+in the last year` $html 1 -}} {{- $total = int (replace (index (index . 0) 1) "," "") -}} {{- else -}} {{- range $cells }}{{ $total = add $total .count }}{{ end -}} {{- end -}} {{- if and $html (not $cells) -}} {{- warnf "contributions: no days found on %s; GitHub may have changed the page" $url -}} {{- end -}} {{- with $cells -}} {{- /* Month labels at the first week of each month; skip one too close to the next or to the end. */ -}} {{- $firstOfWeek := dict -}} {{- range sort $cells "date" -}} {{- $k := string .col -}} {{- if not (index $firstOfWeek $k) }}{{ $firstOfWeek = merge $firstOfWeek (dict $k (.date.Format "Jan")) }}{{ end -}} {{- end -}} {{- $labels := slice -}} {{- $prev := index $firstOfWeek "0" -}} {{- range $i := seq 1 (sub $weeks 1) -}} {{- $m := index $firstOfWeek (string $i) -}} {{- if and $m (ne $m $prev) (le $i (sub $weeks 3)) }}{{ $labels = $labels | append (dict "i" $i "m" $m) }}{{ end -}} {{- $prev = $m -}} {{- end -}}
